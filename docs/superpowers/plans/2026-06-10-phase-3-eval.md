# Phase 3: Eval / Improve Path — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:executing-plans (inline). Steps use `- [ ]`.

**Goal:** Turn a library prompt into a measured, improvable project: `pv init`/`pv promote` create a project; `pv data` curates JSONL datasets (hand-authored + Claude seed-gen); `pv eval` runs a prompt × model over a split with deterministic + LLM-judge scorers, producing a Markdown report and a ledger entry; `pv runs compare` diffs two runs.

**Architecture:** New `datasets/` (Example/Dataset + JSONL io) and `projects/` (Project model + store, `init`/`promote`) modules. `eval/` gains `scorers/` (a registry of deterministic scorers + an `llm_judge`), a `harness` that renders a prompt per example, calls the engine through the existing metering+caching runner, scores, and writes a `Run`, and a `report` (Markdown + compare). `generate/datagen.py` synthesizes seed examples. All engine calls go through the existing `Runner`, so tests use `FakeRunner` with zero spend.

**Tech Stack:** Python 3.13, pydantic v2, typer, rich, pyyaml, jsonschema (new dep for `json_schema_valid`), pytest.

**Decisions (from review):** datasets = hand-authored **and** `pv data gen`; judge model default = **opus** (overridable per-project/CLI); build **full** Phase 3.

```python
from __future__ import annotations
import json
from collections import Counter
from pathlib import Path
from .model import Example

def load_jsonl(path: Path) -> list[Example]:
    path = Path(path)
    if not path.exists():
        return []
    out = []
    for i, line in enumerate(path.read_text().splitlines(), 1):
        line = line.strip()
        if not line:
            continue
        try:
            out.append(Example(**json.loads(line)))
        except Exception as e:
            raise ValueError(f"{path}:{i}: {e}")
    return out

def write_jsonl(path: Path, examples: list[Example]) -> None:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    Path(path).write_text("\n".join(e.model_dump_json() for e in examples) + ("\n" if examples else ""))

def validate(examples: list[Example], required_inputs: list[str] | None = None) -> list[str]:
    problems = []
    ids = [e.id for e in examples]
    dups = [k for k, c in Counter(ids).items() if c > 1]
    if dups:
        problems.append(f"duplicate ids: {dups}")
    for e in examples:
        for f in (required_inputs or []):
            if f not in e.input:
                problems.append(f"{e.id}: missing input field {f!r}")
    return problems

def stats(examples: list[Example]) -> dict:
    keys = Counter(k for e in examples for k in e.input)
    return {
        "count": len(examples),
        "with_reference": sum(1 for e in examples if e.reference is not None),
        "input_fields": dict(keys),
    }
```

```python
from __future__ import annotations
from pathlib import Path
import yaml
from ..prompts.registry import PromptStore
from ..roles.model import load_role
from ..store import paths
from .model import Project

def project_dir(root: Path, name: str) -> Path:
    return paths.projects_dir(root) / name

def project_file(root: Path, name: str) -> Path:
    return project_dir(root, name) / "project.yaml"

def exists(root: Path, name: str) -> bool:
    return project_file(root, name).exists()

def load_project(root: Path, name: str) -> Project:
    f = project_file(root, name)
    if not f.exists():
        raise FileNotFoundError(f"no project named {name!r}")
    return Project(name=name, **(yaml.safe_load(f.read_text()) or {}))

def save_project(root: Path, project: Project) -> None:
    d = project_dir(root, project.name); d.mkdir(parents=True, exist_ok=True)
    body = project.model_dump(exclude={"name"})
    project_file(root, project.name).write_text(yaml.safe_dump(body, sort_keys=False))

def prompt_store_for(root: Path, name: str) -> PromptStore:
    return PromptStore(project_dir(root, name) / "prompts")

def dataset_path(root: Path, project: Project, split: str) -> Path:
    return project_dir(root, project.name) / project.datasets[split]

def init_project(root: Path, name: str, role: str) -> Project:
    role_obj = load_role(role)
    project = Project(name=name, role=role, scorers=role_obj.scorers or ["llm_judge"])
    d = project_dir(root, name); (d / "datasets").mkdir(parents=True, exist_ok=True)
    for split in project.datasets:
        p = dataset_path(root, project, split)
        if not p.exists(): p.write_text("")
    save_project(root, project)
    (d / "README.md").write_text(f"# {name}\n\nRole: {role}. Created by `pv init`.\n")
    return project

def promote(root: Path, prompt_name: str, to_project: str) -> Project:
    lib = PromptStore(paths.library_dir(root))
    pv = lib.load(prompt_name)  # latest
    role = pv.role or "general"
    if exists(root, to_project):
        project = load_project(root, to_project)
    else:
        project = init_project(root, to_project, role)
    prompt_store_for(root, to_project).save_new_version(pv.model_copy(update={"version": 1}))
    return project
```

```python
from __future__ import annotations
import json, re
from jsonschema import Draft202012Validator
from ..result import ScoreResult
from ...datasets.model import Example

class ExactMatch:
    name = "exact_match"
    def score(self, e: Example, output: str) -> ScoreResult:
        ok = output.strip() == str(e.reference).strip()
        return ScoreResult(scorer=self.name, value=1.0 if ok else 0.0, passed=ok,
                           detail="" if ok else f"expected {e.reference!r}")

class Contains:
    name = "contains"
    def __init__(self, needle: str | None = None): self.needle = needle
    def score(self, e: Example, output: str) -> ScoreResult:
        needle = self.needle if self.needle is not None else str(e.reference)
        ok = needle in output
        return ScoreResult(scorer=self.name, value=1.0 if ok else 0.0, passed=ok)

class Regex:
    name = "regex"
    def __init__(self, pattern: str): self.rx = re.compile(pattern)
    def score(self, e: Example, output: str) -> ScoreResult:
        ok = bool(self.rx.search(output))
        return ScoreResult(scorer=self.name, value=1.0 if ok else 0.0, passed=ok)

class JsonSchemaValid:
    name = "json_schema_valid"
    def __init__(self, schema: dict | None = None): self.schema = schema
    def score(self, e: Example, output: str) -> ScoreResult:
        schema = self.schema or e.metadata.get("schema")
        try:
            data = json.loads(output)
        except json.JSONDecodeError as ex:
            return ScoreResult(scorer=self.name, value=0.0, passed=False, detail=f"not JSON: {ex}")
        if schema:
            errs = sorted(Draft202012Validator(schema).iter_errors(data), key=str)
            if errs:
                return ScoreResult(scorer=self.name, value=0.0, passed=False, detail=str(errs[0].message))
        return ScoreResult(scorer=self.name, value=1.0, passed=True)

class NumericTolerance:
    name = "numeric_tolerance"
    def __init__(self, tol: float = 0.0): self.tol = tol
    def score(self, e: Example, output: str) -> ScoreResult:
        m = re.search(r"-?\d+(\.\d+)?", output)
        if not m:
            return ScoreResult(scorer=self.name, value=0.0, passed=False, detail="no number")
        ok = abs(float(m.group()) - float(e.reference)) <= self.tol
        return ScoreResult(scorer=self.name, value=1.0 if ok else 0.0, passed=ok)
```
- [ ] **scorers/llm_judge.py**:
```python
from __future__ import annotations
from ...runners.base import GenParams, Message
from ..result import ScoreResult
from ...datasets.model import Example

JUDGE_SCHEMA = {"type":"object","properties":{
    "score":{"type":"integer"},"passed":{"type":"boolean"},"rationale":{"type":"string"}},
    "required":["score","passed","rationale"],"additionalProperties":False}

DEFAULT_RUBRIC = ("Judge how well the OUTPUT accomplishes the task for the given INPUT"
                  " (and REFERENCE if present). Score 1-5 (5=excellent). passed = score>=4.")

class LLMJudge:
    name = "llm_judge"
    def __init__(self, runner, model: str = "opus", rubric: str = DEFAULT_RUBRIC):
        self.runner = runner; self.model = model; self.rubric = rubric
    def score(self, e: Example, output: str) -> ScoreResult:
        ref = "" if e.reference is None else f"\nREFERENCE:\n{e.reference}"
        msg = (f"{self.rubric}\n\nINPUT:\n{e.input}{ref}\n\nOUTPUT:\n{output}\n\n"
               "Return JSON: score (1-5 int), passed (bool), rationale.")
        resp = self.runner.complete([Message(role="user", content=msg)],
                                    GenParams(model=self.model, json_schema=JUDGE_SCHEMA))
        d = resp.structured or {}
        score = int(d.get("score", 0))
        return ScoreResult(scorer=self.name, value=max(0.0, min(1.0, (score-1)/4)),
                           passed=bool(d.get("passed", score >= 4)),
                           detail=d.get("rationale", ""))
```

```python
from __future__ import annotations
from .deterministic import ExactMatch, Contains, Regex, JsonSchemaValid, NumericTolerance
from .llm_judge import LLMJudge

_DETERMINISTIC = {c.name: c for c in [ExactMatch, Contains, Regex, JsonSchemaValid, NumericTolerance]}

def build_scorers(specs: list, runner=None, judge_model: str = "opus") -> list:
    scorers = []
    for spec in specs:
        if isinstance(spec, str):
            name, cfg = spec, {}
        else:
            cfg = dict(spec); name = cfg.pop("name")
        if name == "llm_judge":
            scorers.append(LLMJudge(runner, model=cfg.get("model", judge_model),
                                    rubric=cfg.get("rubric", LLMJudge.__init__.__defaults__[1])))
        elif name in _DETERMINISTIC:
            scorers.append(_DETERMINISTIC[name](**cfg))
        else:
            raise ValueError(f"unknown scorer: {name}")
    return scorers
```

```python
from __future__ import annotations
import time
from statistics import mean
from ..prompts.model import PromptVersion
from ..runners.base import GenParams, Message
from ..datasets.model import Example
from .result import Run, ExampleResult, ScoreResult

def _run_id(project: str, split: str) -> str:
    return f"{project}-{split}-{int(time.time())}"

def run_eval(runner, prompt: PromptVersion, examples: list[Example], scorers: list,
             model: str = "opus", project: str = "", split: str = "dev") -> Run:
    results: list[ExampleResult] = []
    for e in examples:
        system, user = prompt.render(e.input)
        msgs = [m for m in [Message(role="system", content=system) if system else None,
                            Message(role="user", content=user)] if m]
        before = getattr(runner, "total_cost", None)
        resp = runner.complete(msgs, GenParams(model=model, system=system or None))
        cost = (getattr(runner, "total_cost", 0.0) - before) if before is not None else resp.usage.cost_usd
        scores = [s.score(e, resp.text) for s in scorers]
        results.append(ExampleResult(id=e.id, output=resp.text, scores=scores, cost_usd=cost))
    metrics = _aggregate(results)
    return Run(run_id=_run_id(project, split), project=project, prompt=prompt.name,
               prompt_version=prompt.version, prompt_hash=prompt.content_hash,
               model=getattr(runner, "last_model", "") or model, split=split,
               examples=results, metrics=metrics,
               total_cost_usd=sum(r.cost_usd for r in results), created_at=time.time())

def _aggregate(results: list[ExampleResult]) -> dict:
    by: dict[str, list[ScoreResult]] = {}
    for r in results:
        for s in r.scores:
            by.setdefault(s.scorer, []).append(s)
    return {name: {"mean": round(mean(s.value for s in lst), 4),
                   "pass_rate": round(mean(1.0 if s.passed else 0.0 for s in lst), 4)}
            for name, lst in by.items()}
```

```python
from __future__ import annotations
from .result import Run

def render_report(run: Run) -> str:
    lines = [f"# Eval: {run.project} / {run.prompt} v{run.prompt_version}",
             f"- model: {run.model}  split: {run.split}  examples: {len(run.examples)}",
             f"- run_id: {run.run_id}  cost: ${run.total_cost_usd:.4f}", "", "## Metrics", "",
             "| scorer | mean | pass rate |", "| --- | --- | --- |"]
    for name, m in run.metrics.items():
        lines.append(f"| {name} | {m['mean']:.3f} | {m['pass_rate']:.0%} |")
    lines += ["", "## Examples", "", "| id | scores | output (truncated) |", "| --- | --- | --- |"]
    for r in run.examples:
        sc = " ".join(f"{s.scorer}={s.value:.2f}" for s in r.scores)
        out = r.output.replace("\n", " ")[:80]
        lines.append(f"| {r.id} | {sc} | {out} |")
    return "\n".join(lines) + "\n"

def compare(a: Run, b: Run) -> str:
    scorers = sorted(set(a.metrics) | set(b.metrics))
    lines = [f"# Compare: {a.run_id} vs {b.run_id}", "",
             "| scorer | A mean | B mean | Δ |", "| --- | --- | --- | --- |"]
    for s in scorers:
        am = a.metrics.get(s, {}).get("mean", 0.0); bm = b.metrics.get(s, {}).get("mean", 0.0)
        lines.append(f"| {s} | {am:.3f} | {bm:.3f} | {bm-am:+.3f} |")
    lines += ["", f"- cost: A ${a.total_cost_usd:.4f}  B ${b.total_cost_usd:.4f}"]
    return "\n".join(lines) + "\n"
```

```python
from __future__ import annotations
import json
from ..runners.base import GenParams, Message
from ..prompts.model import PromptVersion
from ..datasets.model import Example

GEN_SCHEMA = {"type":"object","properties":{"examples":{"type":"array","items":{
    "type":"object","properties":{"input_json":{"type":"string"},"reference":{"type":"string"}},
    "required":["input_json"],"additionalProperties":False}}},
    "required":["examples"],"additionalProperties":False}

def gen_examples(runner, prompt: PromptVersion, n: int = 5, model: str = "opus") -> list[Example]:
    variables = sorted(prompt.variables())
    msg = (f"Generate {n} diverse, realistic test examples for this prompt. "
           f"Each example's input_json must be a JSON object with EXACTLY these keys: {variables}. "
           "Provide a concise 'reference' (expected/ideal answer) where it makes sense.\n\n"
           f"PROMPT SYSTEM:\n{prompt.system}\n\nPROMPT USER TEMPLATE:\n{prompt.user}")
    resp = runner.complete([Message(role="user", content=msg)],
                           GenParams(model=model, json_schema=GEN_SCHEMA))
    data = resp.structured or {}
    out = []
    for i, item in enumerate(data.get("examples", []), 1):
        try:
            inp = json.loads(item["input_json"])
        except Exception:
            continue
        out.append(Example(id=f"gen-{i}", input=inp, reference=item.get("reference")))
    return out
```
